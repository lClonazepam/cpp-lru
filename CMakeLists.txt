cmake_minimum_required(VERSION 3.16)
project(cpp_lru LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(test_lru tests/test_lru.cpp)
target_include_directories(test_lru PRIVATE include)
if(MSVC)
  target_compile_options(test_lru PRIVATE /W4)
else()
  target_compile_options(test_lru PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME test_lru COMMAND test_lru)
